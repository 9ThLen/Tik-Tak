include(FetchContent)

FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.15.2
    GIT_SHALLOW    TRUE
)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

add_executable(tiktak_core_tests
    test_fft.cpp
    test_window.cpp
    test_mel.cpp
    test_chroma.cpp
    test_stft.cpp
    test_dft.cpp
    test_logfilt.cpp
    test_resample.cpp
    test_beatnet.cpp
    test_beat_this.cpp
    test_matched.cpp
    test_odf.cpp
    test_scheduler.cpp
    test_click.cpp
    test_metronome.cpp
    test_player.cpp
    test_live_metronome.cpp
    test_particle.cpp
    test_activation_tempo.cpp
    test_live.cpp
    test_octave_freeze.cpp
    test_octave_offset.cpp
    test_bar.cpp
    test_sync.cpp
    test_tempo.cpp
    test_tracker.cpp
    test_downbeat.cpp
    test_offline.cpp
    test_grid_cache.cpp
    test_api.cpp
)

target_link_libraries(tiktak_core_tests PRIVATE tiktak_core GTest::gtest_main)

# Tests reach into the private headers on purpose: the DSP building blocks are
# implementation detail of the C API but still need direct coverage.
target_include_directories(tiktak_core_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../src)

include(GoogleTest)
gtest_discover_tests(tiktak_core_tests)

# Decoding is a separate library with a third-party dependency, so it gets a
# separate test binary — the core suite must stay runnable without it.
if(TIKTAK_BUILD_DECODE)
    add_executable(tiktak_decode_tests test_decode.cpp)
    target_link_libraries(tiktak_decode_tests PRIVATE tiktak_decode GTest::gtest_main)
    target_include_directories(tiktak_decode_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../src)

    # Real encoded files rather than synthesised bytes: the point is to check
    # against what an encoder actually produces, including the padding and
    # delay that a hand-written fixture would quietly omit. Generated by
    # tests/data/generate.py.
    target_compile_definitions(tiktak_decode_tests
        PRIVATE TIKTAK_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")

    gtest_discover_tests(tiktak_decode_tests)
endif()
